;; ===== Sequence.tdl =====
;; ============================================================================
;; Sequence.tdl - permanent company-level barcode sequence
;; Numbers only go UP. Deleted/retired codes are NEVER reused.
;; ============================================================================

[System: UDF]
    ;; Next sequence integer to allocate (starts at 1)
    JGNextSeq             : Number : 21100
    ;; Highest ever issued (audit / never-reuse proof)
    JGHighestIssued       : Number : 21101
    ;; Optional company switch
    JGNativeAddonReady    : Logical : 21102

[Function: JG Ensure Company Sequence]
    01 : IF : $$IsEmpty:$JGNextSeq:Company:##SVCurrentCompany OR $$Number:$JGNextSeq:Company:##SVCurrentCompany < 1
    02 :   New Object : Company : ##SVCurrentCompany
    03 :   Set Value  : JGNextSeq : 1
    04 :   Set Value  : JGHighestIssued : 0
    05 :   Set Value  : JGNativeAddonReady : Yes
    06 :   Accept Alter
    07 : ENDIF

[Function: JG Pad Number]
    ;; Returns zero-padded string of length @@JGBarcodePadLength
    Parameter : pNum : Number
    Variable  : S : String
    Variable  : Need : Number
    01 : Set : S : $$String:##pNum
    02 : Set : Need : @@JGBarcodePadLength - $$StringLength:##S
    03 : WHILE : ##Need > 0
    04 :   Set : S : "0" + ##S
    05 :   Set : Need : ##Need - 1
    06 : END WHILE
    07 : RETURN : ##S

[Function: JG Format Internal Barcode]
    Parameter : pNum : Number
    01 : RETURN : @@JGBarcodePrefix + $$Call:JG Pad Number:##pNum

[Function: JG Allocate Next Barcode]
    ;; Atomically (within Tally object accept) allocate next JG code.
    ;; Never reuses previous numbers.
    Variable : Seq : Number
    Variable : Code : String
    01 : Call : JG Ensure Company Sequence
    02 : Set : Seq : $$Number:$JGNextSeq:Company:##SVCurrentCompany
    03 : IF : ##Seq < 1
    04 :   Set : Seq : 1
    05 : ENDIF
    06 : Set : Code : $$Call:JG Format Internal Barcode:##Seq
    07 : New Object : Company : ##SVCurrentCompany
    08 : Set Value  : JGNextSeq : ##Seq + 1
    09 : Set Value  : JGHighestIssued : ##Seq
    10 : Accept Alter
    11 : Set : JGWorkBarcode : ##Code
    12 : RETURN : ##Code

[Function: JG Peek Next Barcode]
    Variable : Seq : Number
    01 : Call : JG Ensure Company Sequence
    02 : Set : Seq : $$Number:$JGNextSeq:Company:##SVCurrentCompany
    03 : IF : ##Seq < 1
    04 :   Set : Seq : 1
    05 : ENDIF
    06 : RETURN : $$Call:JG Format Internal Barcode:##Seq

;; ===== Code128.tdl =====
;; ============================================================================
;; Code128.tdl - Code 128 Subset B encoder for Libre Barcode 128 font
;;
;; Install font:  fonts\LibreBarcode128-Regular.ttf  (double-click -> Install)
;; Font family name used in reports:  Libre Barcode 128
;;
;; Encoded pattern for Libre Barcode 128:
;;   Start B glyph + data glyphs + checksum glyph + Stop glyph
;; Mapping uses Latin-1 / Windows-1252 style codes commonly required by
;; the Libre Barcode 128 font (code point = value + 100 for codes 0-94,
;; with Start B = 204, Stop = 206 in many Tally deployments using this font).
;; ============================================================================

[Function: JG Code128 Char Value]
    ;; Subset B: ASCII 32..126 -> value 0..94
    Parameter : pChar : String
    Variable  : A : Number
    01 : Set : A : $$ASCIICode:##pChar
    02 : IF : ##A < 32 OR ##A > 126
    03 :   RETURN : 0
    04 : ENDIF
    05 : RETURN : ##A - 32

[Function: JG Code128 Glyph From Value]
    ;; Map code value 0..106 to a printable glyph for Libre Barcode 128
    Parameter : pVal : Number
    01 : IF : ##pVal = 103
    02 :   RETURN : $$Char:203
    03 : ENDIF
    04 : IF : ##pVal = 104
    05 :   RETURN : $$Char:204
    06 : ENDIF
    07 : IF : ##pVal = 105
    08 :   RETURN : $$Char:205
    09 : ENDIF
    10 : IF : ##pVal = 106
    11 :   RETURN : $$Char:206
    12 : ENDIF
    13 : IF : ##pVal >= 0 AND ##pVal <= 94
    14 :   RETURN : $$Char:(##pVal + 32)
    15 : ENDIF
    16 : ;; Fallback for 95..102 (rarely needed for alphanumeric JG barcodes)
    17 : RETURN : $$Char:(##pVal + 100)

[Function: JG Code128 Encode]
    ;; Full Code128-B encode of human-readable barcode text
    Parameter : pText : String
    Variable  : i : Number
    Variable  : Len : Number
    Variable  : Ch : String
    Variable  : Val : Number
    Variable  : Sum : Number
    Variable  : Pos : Number
    Variable  : Out : String
    Variable  : Check : Number

    01 : IF : $$IsEmpty:##pText
    02 :   RETURN : ""
    03 : ENDIF

    04 : Set : Len : $$StringLength:##pText
    05 : Set : Sum : 104
    06 : Set : Out : $$Call:JG Code128 Glyph From Value:104
    07 : Set : i : 1
    08 : Set : Pos : 1

    09 : WHILE : ##i <= ##Len
    10 :   Set : Ch  : $$StringPart:##pText:##i:1
    11 :   Set : Val : $$Call:JG Code128 Char Value:##Ch
    12 :   Set : Sum : ##Sum + (##Val * ##Pos)
    13 :   Set : Out : ##Out + $$Call:JG Code128 Glyph From Value:##Val
    14 :   Set : Pos : ##Pos + 1
    15 :   Set : i   : ##i + 1
    16 : END WHILE

    17 : Set : Check : $$Round:##Sum:0 - ($$Round:(##Sum / 103):0 * 103)
    ;; Mod 103 without modulo operator: Check = Sum - floor(Sum/103)*103
    18 : IF : ##Check < 0
    19 :   Set : Check : ##Check + 103
    20 : ENDIF
    21 : IF : ##Check > 102
    22 :   Set : Check : 0
    23 : ENDIF

    24 : Set : Out : ##Out + $$Call:JG Code128 Glyph From Value:##Check
    25 : Set : Out : ##Out + $$Call:JG Code128 Glyph From Value:106
    26 : Set : JGWorkEncoded : ##Out
    27 : RETURN : ##Out

;; Safer mod helper used if Round-based mod misbehaves on some builds
[Function: JG Mod103]
    Parameter : pSum : Number
    Variable  : Q : Number
    Variable  : R : Number
    01 : Set : Q : $$Number:$$String:$$LocaleString:(##pSum / 103):"0"
    ;; Prefer integer division via repeated subtract for reliability
    02 : Set : R : ##pSum
    03 : WHILE : ##R >= 103
    04 :   Set : R : ##R - 103
    05 : END WHILE
    06 : WHILE : ##R < 0
    07 :   Set : R : ##R + 103
    08 : END WHILE
    09 : RETURN : ##R

[Function: JG Code128 Encode Safe]
    Parameter : pText : String
    Variable  : i : Number
    Variable  : Len : Number
    Variable  : Ch : String
    Variable  : Val : Number
    Variable  : Sum : Number
    Variable  : Pos : Number
    Variable  : Out : String
    Variable  : Check : Number
    01 : IF : $$IsEmpty:##pText
    02 :   RETURN : ""
    03 : ENDIF
    04 : Set : Len : $$StringLength:##pText
    05 : Set : Sum : 104
    06 : Set : Out : $$Call:JG Code128 Glyph From Value:104
    07 : Set : i : 1
    08 : Set : Pos : 1
    09 : WHILE : ##i <= ##Len
    10 :   Set : Ch  : $$StringPart:##pText:##i:1
    11 :   Set : Val : $$Call:JG Code128 Char Value:##Ch
    12 :   Set : Sum : ##Sum + (##Val * ##Pos)
    13 :   Set : Out : ##Out + $$Call:JG Code128 Glyph From Value:##Val
    14 :   Set : Pos : ##Pos + 1
    15 :   Set : i   : ##i + 1
    16 : END WHILE
    17 : Set : Check : $$Call:JG Mod103:##Sum
    18 : Set : Out : ##Out + $$Call:JG Code128 Glyph From Value:##Check
    19 : Set : Out : ##Out + $$Call:JG Code128 Glyph From Value:106
    20 : Set : JGWorkEncoded : ##Out
    21 : RETURN : ##Out

    Parameter : pItem : String
    Parameter : pBarcode : String
    Parameter : pMfr : String
    Variable  : Enc : String
    01 : IF : $$IsEmpty:##pItem OR $$IsEmpty:##pBarcode
    02 :   RETURN : No

    05 : New Object : Stock Item : ##pItem
    06 : Set Value  : JGBarcode : ##pBarcode

    09 :   Set Value : JGManufacturerBarcode : ##pMfr
    10 : ENDIF
    11 : Set Value : JGBarcodeAssignedOn : $$MachineDate
    12 : Accept Alter
    13 : RETURN : Yes

;; ===== PurchaseEvents.tdl =====
;; ============================================================================
;; PurchaseEvents.tdl - On Purchase voucher save:
;;   - If Manufacturer Barcode exists -> store it
;;   - If internal Barcode empty -> allocate JG######## (never reused)
;;   - Prompt: Print Barcode Labels?
;;   - Print 50x25mm labels via Tally (Windows / thermal driver)
;; ============================================================================

[Function: JG Native On Voucher Accept]
    01 : IF : NOT @@JGBarcodeEnabled
    02 :   RETURN : Yes
    03 : ENDIF
    04 : IF : $$IsPurchase:$VoucherTypeName OR $$IsEqual:$VoucherTypeName:"Purchase" OR $$IsEqual:$VoucherTypeName:"Purchase Order"
    05 :   Call : JG Native Process Purchase
    06 : ENDIF
    07 : RETURN : Yes

[Function: JG Native Process Purchase]
    Variable : ItemName : String
    Variable : Existing : String
    Variable : Mfr : String
    Variable : NewCode : String
    Variable : Qty : Number
    Variable : AnyDone : Logical
    Variable : LabelBlob : String

    01 : Call : JG Ensure Company Sequence
    02 : Set : AnyDone : No
    03 : Set : LabelBlob : ""
    04 : Set : JGLabelList : ""

    05 : Walk Collection : Inventory Entries
    06 :   Set : ItemName : $StockItemName
    07 :   IF : NOT $$IsEmpty:##ItemName
    08 :     Set : Existing : $JGBarcode:StockItem:##ItemName
    09 :     Set : Mfr : $JGManufacturerBarcode:StockItem:##ItemName
    10 :     IF : $$IsEmpty:##Mfr
    11 :       Set : Mfr : ""
    12 :     ENDIF
    13 :     IF : $$IsEmpty:##Existing
    14 :       Set : NewCode : $$Call:JG Allocate Next Barcode
    15 :       Call : JG Persist Item Barcode : ##ItemName : ##NewCode : ##Mfr
    16 :       Set : Existing : ##NewCode
    17 :       Set : AnyDone : Yes
    18 :     ELSE
    19 :       ;; Keep permanent barcode; still refresh manufacturer if provided on master
    20 :       IF : NOT $$IsEmpty:##Mfr
    21 :         New Object : Stock Item : ##ItemName
    22 :         Set Value  : JGManufacturerBarcode : ##Mfr
    23 :         Accept Alter
    24 :       ENDIF
    25 :       ;; Ensure encoded string exists for printing
    26 :       IF : $$IsEmpty:$JGBarcodeEncoded:StockItem:##ItemName
    27 :         Call : JG Persist Item Barcode : ##ItemName : ##Existing : ##Mfr
    28 :       ENDIF
    29 :     ENDIF
    30 :     Set : Qty : $$Number:$BilledQty
    31 :     IF : ##Qty < 1
    32 :       Set : Qty : 1
    33 :     ENDIF
    34 :     ;; Pipe-delimited print queue: Item|Copies~
    35 :     Set : LabelBlob : ##LabelBlob + ##ItemName + "|" + $$String:##Qty + "~"
    36 :   ENDIF
    37 : End Walk

    38 : Set : JGLabelList : ##LabelBlob

    39 : IF : $$IsEmpty:##LabelBlob
    40 :   RETURN
    41 : ENDIF

    42 : IF : @@JGAskPrintOnPurchase
    43 :   Query Box : "Print Barcode Labels?" : JGPrintLabelsChoice
    44 :   IF : ##JGPrintLabelsChoice
    45 :     Call : JG Print Label Queue
    46 :   ENDIF
    47 : ENDIF

;; ===== SalesScan.tdl =====
;; ============================================================================
;; SalesScan.tdl - scanner support during sales (no external service)
;; Alt+B -> scan -> stock item selected -> Qty 1 -> Enter completes invoice
;; ============================================================================

[Key: JG Native Sales Scan Key]
    Key      : Alt + B
    Action   : Call : JG Native Sales Scan
    Inactive : NOT ($$IsSales:$VoucherTypeName OR $$IsEqual:$VoucherTypeName:"Sales")

[Function: JG Native Sales Scan]
    Variable : Scanned : String
    Variable : Item : String
    01 : Input Value : "Scan Barcode" : JGScanBuffer : String
    02 : Set : Scanned : ##JGScanBuffer
    03 : IF : $$IsEmpty:##Scanned
    04 :   RETURN
    05 : ENDIF

    06 : Set : Item : $$FilterValue:$Name:StockItem:JG By Barcode Coll:##Scanned
    07 : IF : $$IsEmpty:##Item
    08 :   Set : Item : $$FilterValue:$Name:StockItem:JG By Mfr Coll:##Scanned
    09 : ENDIF

    10 : IF : $$IsEmpty:##Item
    11 :   MSG BOX : "Barcode Scan" : "No stock item for: " + ##Scanned
    12 :   RETURN
    13 : ENDIF

    14 : New Object : Inventory Entries
    15 : Set Value  : StockItemName : ##Item
    16 : Set Value  : BilledQty : 1
    17 : IF : NOT $$IsEmpty:$JGSellingRate:StockItem:##Item
    18 :   Set Value : Rate : $JGSellingRate:StockItem:##Item
    19 : ENDIF
    20 : Accept Alter
    21 : MSG BOX : "Barcode Scan" : ##Item + "  |  Qty 1  |  Press Enter to continue"

[Function: JG Resolve Barcode To Item]
    Parameter : pCode : String
    Variable  : Item : String
    01 : Set : Item : $$FilterValue:$Name:StockItem:JG By Barcode Coll:##pCode
    02 : IF : $$IsEmpty:##Item
    03 :   Set : Item : $$FilterValue:$Name:StockItem:JG By Mfr Coll:##pCode
    04 : ENDIF
    05 : RETURN : ##Item

;; ===== LabelPrint.tdl =====
;; ============================================================================
;; LabelPrint.tdl - 50mm x 25mm native Tally label (Code128 font)
;; Kept to attributes TallyPrime accepts on Form/Part/Field/Report
;; ============================================================================

[Function: JG Build Print Collection From Queue]
    01 : IF : $$IsEmpty:##JGLabelList
    02 :   RETURN : No
    03 : ENDIF
    04 : RETURN : Yes

[Function: JG Print Label Queue]
    Variable : Queue : String
    Variable : Piece : String
    Variable : Sep   : Number
    Variable : Item  : String
    Variable : Copies : Number
    Variable : Pipe  : Number
    Variable : c : Number

    01 : IF : $$IsEmpty:##JGLabelList
    02 :   MSG BOX : "JG Barcode" : "No labels in print queue."
    03 :   RETURN
    04 : ENDIF

    05 : Set : Queue : ##JGLabelList

    06 : WHILE : NOT $$IsEmpty:##Queue
    07 :   Set : Sep : $$StringFind:##Queue:"~"
    08 :   IF : ##Sep < 1
    09 :     Set : Piece : ##Queue
    10 :     Set : Queue : ""
    11 :   ELSE
    12 :     Set : Piece : $$StringPart:##Queue:1:(##Sep - 1)
    13 :     Set : Queue : $$StringPart:##Queue:(##Sep + 1):$$StringLength:##Queue
    14 :   ENDIF

    15 :   IF : NOT $$IsEmpty:##Piece
    16 :     Set : Pipe : $$StringFind:##Piece:"|"
    17 :     IF : ##Pipe < 1
    18 :       Set : Item : ##Piece
    19 :       Set : Copies : 1
    20 :     ELSE
    21 :       Set : Item : $$StringPart:##Piece:1:(##Pipe - 1)
    22 :       Set : Copies : $$Number:$$StringPart:##Piece:(##Pipe + 1):$$StringLength:##Piece
    23 :     ENDIF
    24 :     IF : ##Copies < 1
    25 :       Set : Copies : 1
    26 :     ENDIF

    27 :     Set : JGWorkItem : ##Item
    28 :     Set : JGWorkBarcode : $JGBarcode:StockItem:##Item
    29 :     Set : JGWorkEncoded : $JGBarcodeEncoded:StockItem:##Item
    30 :     Set : JGWorkBrand : $JGBrand:StockItem:##Item
    31 :     Set : JGWorkMrp : $JGMrp:StockItem:##Item
    32 :     Set : JGWorkSellRate : $JGSellingRate:StockItem:##Item

    33 :     IF : NOT $$IsEmpty:##JGWorkBarcode
    34 :       IF : $$IsEmpty:##JGWorkEncoded
    35 :         Set : JGWorkEncoded : $$Call:JG Code128 Encode Safe:##JGWorkBarcode
    36 :       ENDIF
    37 :       Set : c : 1
    38 :       WHILE : ##c <= ##Copies
    39 :         Display : JG Barcode Label Report
    40 :         Set : c : ##c + 1
    41 :       END WHILE
    42 :     ENDIF
    43 :   ENDIF
    44 : END WHILE

    45 : MSG BOX : "JG Barcode" : "Label preview/print complete. Use Print (Ctrl+P) on the label screen if needed."

    03 :   Set : Item : $$FilterValue:$Name:StockItem:JG By Mfr Coll:##pCode

[Function: JG Tools Search]
    Variable : Item : String
    01 : Input Value : "Scan / type barcode" : JGScanBuffer : String
    02 : Set : Item : $$Call:JG Resolve Barcode To Item:##JGScanBuffer
    03 : IF : $$IsEmpty:##Item
    04 :   MSG BOX : "Search" : "Not found"
    05 : ELSE
    06 :   MSG BOX : "Search" : "Item: " + ##Item + $$LocalCRLF + "Barcode: " + $JGBarcode:StockItem:##Item + $$LocalCRLF + "Mfr: " + $JGManufacturerBarcode:StockItem:##Item + $$LocalCRLF + "MRP: " + $$String:$JGMrp:StockItem:##Item + $$LocalCRLF + "Rate: " + $$String:$JGSellingRate:StockItem:##Item
    07 : ENDIF

[Function: JG Tools Batch Print]
    Variable : Item : String
    Variable : Copies : String
    01 : Input Value : "Stock Item Name" : JGWorkItem : String
    02 : Set : Item : ##JGWorkItem
    03 : IF : $$IsEmpty:##Item
    04 :   RETURN
    05 : ENDIF
    06 : IF : $$IsEmpty:$JGBarcode:StockItem:##Item
    07 :   MSG BOX : "Batch Print" : "Item has no barcode. Save a Purchase first or assign barcode."
    08 :   RETURN
    09 : ENDIF
    10 : Input Value : "Number of copies" : JGScanBuffer : String
    11 : Set : Copies : ##JGScanBuffer
    12 : IF : $$IsEmpty:##Copies
    13 :   Set : Copies : "1"
    14 : ENDIF
    15 : Set : JGLabelList : ##Item + "|" + ##Copies + "~"
    16 : Call : JG Print Label Queue
